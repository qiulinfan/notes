> \begin{definition}{\kn{moment generating function}}

>
> \[\mathit{M}_{\mathit{X}}(\mathit{t}) = \mathbb{E}\lbrack\mathit{e}^{\mathit{t}\mathit{X}}\rbrack,\quad\mathit{t} \in \mathbb{R}\]

> \end{definition}
