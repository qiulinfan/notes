> \begin{proposition}{\kn{absolutely continuous 的性质}}
>
> \begin{itemize}
> \item
>   对 \knref{total variation measure},
>

>
>   (容易证明)

>   \[\nu \perp \mu\ \text{and}\ \nu \ll \mu\Longrightarrow\nu = 0\]

>   (刚才已经证明)
> \end{itemize}

> \end{proposition}
